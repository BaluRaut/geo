4
src/index.ts — Hono app
📄 src/index.ts
import { Hono }              from 'hono'
import { handle }            from 'hono/aws-lambda'
import { cors }              from 'hono/cors'
import { jwt, sign }        from 'hono/jwt'
import { getSignedCookies } from '@aws-sdk/cloudfront-signer'

const app = new Hono()

// ── CORS ─────────────────────────────────────────────────────────
app.use('*', cors({
  origin:       process.env.ALLOWED_ORIGIN ?? '*',
  credentials:  true,
  allowHeaders: ['Authorization', 'Content-Type'],
}))

// ── PUBLIC: exchange API key → JWT ───────────────────────────────
app.post('/auth/token', async (c) => {
  const { apiKey } = await c.req.json<{ apiKey: string }>()
  const valid = (process.env.API_KEYS ?? '')
    .split(',').map(k => k.trim())

  if (!apiKey || !valid.includes(apiKey))
    return c.json({ error: 'Invalid API key' }, 401)

  const token = await sign(
    {
      sub: 'geo-client',
      iat: Math.floor(Date.now() / 1000),
      exp: Math.floor(Date.now() / 1000) + 60 * 60 * 8,
    },
    process.env.JWT_SECRET!
  )

  return c.json({ token, expiresIn: 28800 })
})

// ── PROTECTED: all /api/* require Bearer JWT ─────────────────────
app.use('/api/*', jwt({ secret: process.env.JWT_SECRET! }))

// GET /api/tiles/auth → CloudFront signed cookies
app.get('/api/tiles/auth', async (c) => {
  const expiry = Math.floor(Date.now() / 1000) + 60 * 60 * 8

  const cookies = getSignedCookies({
    keyPairId:  process.env.CF_KEY_PAIR_ID!,
    privateKey: process.env.CF_PRIVATE_KEY!,
    policy: JSON.stringify({
      Statement: [{
        Resource:  `https://${process.env.CF_DOMAIN}/fgb/*`,
        Condition: { DateLessThan: { 'AWS:EpochTime': expiry } },
      }],
    }),
  })

  c.header('Set-Cookie', `CloudFront-Policy=${cookies['CloudFront-Policy']}; Path=/; Secure; SameSite=None`)
  c.header('Set-Cookie', `CloudFront-Signature=${cookies['CloudFront-Signature']}; Path=/; Secure; SameSite=None`)
  c.header('Set-Cookie', `CloudFront-Key-Pair-Id=${cookies['CloudFront-Key-Pair-Id']}; Path=/; Secure; SameSite=None`)
  return c.json({ ok: true })
})

// health check
app.get('/health', (c) => c.json({ status: 'ok' }))

export const handler = handle(app)
5
Store secrets in AWS SSM
Run once before deploy. These are read at Lambda runtime via ${ssm:...} in serverless.yml.
# JWT signing secret (random 32-byte base64)
aws ssm put-parameter --name /geo/jwt-secret \
  --type SecureString \
  --value "$(openssl rand -base64 32)"

# Pre-shared API keys your React app will send
aws ssm put-parameter --name /geo/api-keys \
  --type SecureString \
  --value "key-abc123,key-xyz789"

# CloudFront key pair (from AWS console → Security Credentials)
aws ssm put-parameter --name /geo/cf-key-pair-id \
  --type String --value "APKAXXXXXXXXXX"

aws ssm put-parameter --name /geo/cf-private-key \
  --type SecureString \
  --value "$(cat cloudfront-private-key.pem)"

aws ssm put-parameter --name /geo/cf-domain \
  --type String --value "d1234abcd.cloudfront.net"
⚠️ The CloudFront private key is from AWS Console → Account menu → Security Credentials → CloudFront key pairs → Create. Download it once — AWS won't let you re-download it.

React — src/hooks/useAuth.ts
Token is stored in memory (not localStorage) so it's cleared on refresh. Auto-fires an event when JWT expires so the app can prompt re-login.
📄 src/hooks/useAuth.ts
const API = import.meta.env.VITE_API_URL

let _token: string | null = null          // in-memory only
let _expiry: number  = 0

export async function login(apiKey: string) {
  const res = await fetch(`${API}/auth/token`, {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify({ apiKey }),
  })
  if (!res.ok) throw new Error('Auth failed')

  const { token, expiresIn } = await res.json()
  _token  = token
  _expiry = Date.now() + expiresIn * 1000 - 60_000   // 1 min early refresh
  return token
}

export function isLoggedIn() {
  return !!_token && Date.now() < _expiry
}

// Authenticated fetch — attaches Bearer, refreshes on 401
export async function apiFetch(path: string, init?: RequestInit) {
  const res = await fetch(`${API}${path}`, {
    ...init,
    credentials: 'include',
    headers: {
      ...init?.headers,
      'Authorization': `Bearer ${_token}`,
    },
  })

  if (res.status === 401) {
    _token = null
    window.dispatchEvent(new Event('auth:expired'))
  }

  return res
}
8
React — wire auth into map component
Replace the bare fetch('/fgb/...') with auth-aware loading. One useEffect on mount handles login + cookie fetch before any tile requests fire.
📄 src/UkraineLayerMap.tsx — add at top of component
import { login, apiFetch, isLoggedIn } from './hooks/useAuth'

// inside UkraineLayerMap(), before other useEffects:
useEffect(() => {
  async function initAuth() {
    if (!isLoggedIn())
      await login(import.meta.env.VITE_API_KEY)

    await apiFetch('/api/tiles/auth')
  }
  initAuth()

  // re-login if JWT expires mid-session
  const onExpired = async () => {
    await login(import.meta.env.VITE_API_KEY)
    await apiFetch('/api/tiles/auth')
  }
  window.addEventListener('auth:expired', onExpired)
  return () => window.removeEventListener('auth:expired', onExpired)
}, [])
